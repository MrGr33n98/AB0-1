PASSO 1: ESCOLHA UMA OPÇÃO (faça na VM via SSH)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

OPÇÃO A - Criar usuário no banco existente (RECOMENDADO - mantém dados):
─────────────────────────────────────────────────────────────────────────

ssh sua-vm

# Criar o usuário 'ab0'
docker exec ab0-postgres psql -U postgres -c "CREATE USER ab0 WITH PASSWORD 'ZAbgbZeVAK!5!' CREATEDB SUPERUSER;"

# Criar o banco
docker exec ab0-postgres psql -U postgres -c "CREATE DATABASE ab0_production OWNER ab0;"

# Garantir privilégios
docker exec ab0-postgres psql -U postgres -c "GRANT ALL PRIVILEGES ON DATABASE ab0_production TO ab0;"

# ✅ Verificar se funcionou
docker exec ab0-postgres psql -U ab0 -d ab0_production -c "SELECT version();"

# Se deu certo, você verá a versão do PostgreSQL!


OPÇÃO B - Recriar banco do zero (⚠️ APAGA TODOS OS DADOS):
──────────────────────────────────────────────────────────────

ssh sua-vm
cd $HOME/ab0-app

# CUIDADO: Isso remove TODOS os dados!
docker-compose down -v

# Subir novamente
docker-compose up -d db

# Aguardar inicialização
sleep 15

# Verificar
docker exec ab0-postgres psql -U ab0 -d ab0_production -c "SELECT version();"
