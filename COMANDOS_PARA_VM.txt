VOCÊ ESCOLHEU: OPÇÃO A - Criar usuário mantendo dados existentes ✅

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PASSO 1: Fazer SSH na VM
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ssh seu-usuario@seu-servidor

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PASSO 2: Executar os comandos (copie e cole TUDO de uma vez)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Criar usuário 'ab0'
docker exec ab0-postgres psql -U postgres -c "CREATE USER ab0 WITH PASSWORD 'ZAbgbZeVAK!5!' CREATEDB SUPERUSER;"

# Criar banco 'ab0_production'
docker exec ab0-postgres psql -U postgres -c "CREATE DATABASE ab0_production OWNER ab0;"

# Garantir privilégios
docker exec ab0-postgres psql -U postgres -c "GRANT ALL PRIVILEGES ON DATABASE ab0_production TO ab0;"

# Testar conexão
docker exec ab0-postgres psql -U ab0 -d ab0_production -c "SELECT current_user, current_database(), version();"

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✅ Isso significa que funcionou!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SE DER ERRO "already exists":
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Não tem problema! Significa que já existe. Execute estes comandos para atualizar:

# Atualizar senha do usuário
docker exec ab0-postgres psql -U postgres -c "ALTER USER ab0 WITH PASSWORD 'ZAbgbZeVAK!5!' CREATEDB SUPERUSER;"

# Atualizar owner do banco
docker exec ab0-postgres psql -U postgres -c "ALTER DATABASE ab0_production OWNER TO ab0;"

# Garantir privilégios novamente
docker exec ab0-postgres psql -U postgres -c "GRANT ALL PRIVILEGES ON DATABASE ab0_production TO ab0;"

# Testar
docker exec ab0-postgres psql -U ab0 -d ab0_production -c "SELECT version();"

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

VERIFICAÇÕES ADICIONAIS:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Listar todos os usuários
docker exec ab0-postgres psql -U postgres -c "\du"

# Listar todos os bancos
docker exec ab0-postgres psql -U postgres -c "\l"

# Ver logs do PostgreSQL
docker logs ab0-postgres --tail 50

# Reiniciar containers se necessário
cd $HOME/ab0-app
docker-compose restart
